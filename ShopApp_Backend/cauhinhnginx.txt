server {
    server_name vps.luxuryshop.x10.mx;

    location / {
        proxy_pass http://127.0.0.1:4200;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

location /api/ {
        proxy_pass http://127.0.0.1:8099;  # trỏ đến backend container
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection keep-alive;
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/vps.luxuryshop.x10.mx/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/vps.luxuryshop.x10.mx/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}

server {
    if ($host = vps.luxuryshop.x10.mx) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name vps.luxuryshop.x10.mx;
    return 404; # managed by Certbot


}
